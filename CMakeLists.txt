cmake_minimum_required(VERSION 3.20)
project(SurfaceInspection VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs ml videoio geometry)
find_package(OpenGL REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
find_package(GLEW REQUIRED)
find_package(Threads REQUIRED)
set(IMGUI_DIR "${CMAKE_SOURCE_DIR}/.deps/imgui-1.91.9b" CACHE PATH "Dear ImGui source")
# Dear ImGui uses the same GLEW function loader as our renderer. Disable its embedded loader.
set_source_files_properties(${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp
  PROPERTIES COMPILE_DEFINITIONS IMGUI_IMPL_OPENGL_LOADER_CUSTOM)
if(MSVC)
  set_property(SOURCE ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp
    APPEND PROPERTY COMPILE_OPTIONS /FIGL/glew.h)
else()
  set_property(SOURCE ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp
    APPEND PROPERTY COMPILE_OPTIONS -include GL/glew.h)
endif()
add_library(analysis src/pipeline.cpp src/evaluation.cpp)
target_include_directories(analysis PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(analysis PUBLIC ${OpenCV_LIBS})
if(WIN32)
  target_link_libraries(analysis PUBLIC bcrypt)
endif()
add_library(graphics src/geometry.cpp)
target_include_directories(graphics PUBLIC src)
add_executable(inspect src/main.cpp src/viewer.cpp src/integration.cpp
  ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp
  ${IMGUI_DIR}/backends/imgui_impl_glfw.cpp ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(inspect PRIVATE ${IMGUI_DIR} ${IMGUI_DIR}/backends)
target_link_libraries(inspect PRIVATE analysis graphics glfw GLEW::GLEW OpenGL::GL Threads::Threads)
if(WIN32)
  target_link_libraries(inspect PRIVATE comdlg32)
endif()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE analysis graphics Threads::Threads)
enable_testing()
add_test(NAME core_tests COMMAND core_tests)
if(MINGW)
  target_compile_options(analysis PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(inspect PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
